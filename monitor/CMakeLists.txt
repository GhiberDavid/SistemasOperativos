cmake_minimum_required(VERSION 3.16)
project(monitor_procesos CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

option(MONITOR_BUILD_TUI "Interfaz de terminal (FTXUI)" ON)
option(MONITOR_BUILD_GUI "Interfaz grafica (raylib)" ON)
option(MONITOR_FAKE_OS   "Backend simulado (solo para pruebas fuera de Windows)" OFF)

# ---------------- core: logica compartida, sin librerias de interfaz ----------------
set(CORE_SRC core/model.cpp core/monitor.cpp core/session.cpp)
if(WIN32 AND NOT MONITOR_FAKE_OS)
  list(APPEND CORE_SRC core/os_win.cpp)
else()
  list(APPEND CORE_SRC tests/fake_os.cpp)
endif()
add_library(core STATIC ${CORE_SRC})
target_include_directories(core PUBLIC core)
if(WIN32 AND NOT MONITOR_FAKE_OS)
  target_link_libraries(core PUBLIC psapi advapi32 user32)
endif()
find_package(Threads REQUIRED)
target_link_libraries(core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(core PRIVATE /utf-8)
endif()

include(FetchContent)

# ---------------- TUI (FTXUI) ----------------
if(MONITOR_BUILD_TUI)
  set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(ftxui GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git GIT_TAG v6.1.9 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(ftxui)   # descarga FTXUI (necesita git e internet la primera vez)
  add_executable(monitor_tui tui/main.cpp)
  target_link_libraries(monitor_tui PRIVATE core ftxui::screen ftxui::dom ftxui::component)
  if(WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_link_options(monitor_tui PRIVATE -static)   # sin DLL de MSYS2: el .exe corre desde PowerShell
  endif()
endif()

# ---------------- GUI (raylib) ----------------
if(MONITOR_BUILD_GUI)
  find_package(raylib QUIET)
  if(NOT raylib_FOUND)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(raylib)
  endif()
  add_executable(monitor_gui gui/main.cpp)
  target_link_libraries(monitor_gui PRIVATE core raylib)
  if(WIN32)
    target_link_libraries(monitor_gui PRIVATE opengl32 gdi32 winmm)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_link_options(monitor_gui PRIVATE -static)
    endif()
  endif()
endif()

# ---------------- pruebas del core (con backend simulado) ----------------
if(NOT WIN32 OR MONITOR_FAKE_OS)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE core)
endif()
